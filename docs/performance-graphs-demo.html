🔧 Integration (5 steps, ~15 min)

// 1. Import and initialize PerformanceLogger
import { PerformanceLogger } from './core/PerformanceLogger';

const perfTracker = new PerformanceTracker(surfaceType);
const debugOverlay = new DebugOverlay(perfTracker);
const perfLogger = new PerformanceLogger(surfaceType); // ADD THIS

// 2. Update game loop to track enemy types
const enemyTypes = new Map();
for (const enemy of enemySpawner.getEnemies()) {
  const type = enemy.constructor.name.toLowerCase();
  enemyTypes.set(type, (enemyTypes.get(type) || 0) + 1);
}

perfTracker.setEntityCount(enemyCount);
perfTracker.setBulletCount(bulletCount);
perfTracker.setEnemyTypes(enemyTypes); // ADD THIS
perfTracker.recordFrame(frameDt);

perfLogger.setFrameData(perfTracker.fps, enemyCount, bulletCount); // ADD THIS
perfLogger.setEnemyTypes(enemyTypes); // ADD THIS
perfLogger.recordFrame(frameDt); // ADD THIS

// 3. Pass logger to PauseMenu
pauseMenu.setPerformanceLogger(perfLogger); // ADD THIS

// 4. Save session on game end
perfTracker.saveSession();
const shouldShowReport = perfLogger.saveSession(); // ADD THIS
if (shouldShowReport) {
  console.log('10 games completed! Performance report available.');
}

⚡ Performance Characteristics

  • Zero-GC ring buffer - Pre-allocated 3600-element array
  • ~50KB bundle size - No external charting libraries
  • Lazy-loaded - Only imported when modal opens
  • Canvas 2D rendering - Optimized for 3600 data points
  • localStorage persistence - Last 10 sessions (~5MB)
